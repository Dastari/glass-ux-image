import { effects } from 'glass-ux-image/tools/effects';
import type { EditorSession } from 'glass-ux-image/session';

export const tools = [effects];

// One undoable step, which onAction can see and refuse.
export function addPosterise(editor: EditorSession) {
  editor.run('effects.add', {
    label: 'Posterise',
    entry: {
      id: 'posterise-1',
      filter: 'posterise',
      params: { amount: 1, radius: 2, levels: 6 },
      strength: 0.8,
    },
  });
}
