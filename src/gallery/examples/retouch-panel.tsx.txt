import { type ReactNode } from "react";
import { SectionPage, Specimen } from "glass-ux-docs-kit/react";
import { ImageCanvas } from "glass-ux-image/canvas";
import { ImageEditorProvider, useImageEditor, useSessionState } from "glass-ux-image/react";
import type { EditorSession } from "glass-ux-image/session";
import { RetouchOverlay, RetouchPanel } from "glass-ux-image/tools/retouch";
import { brush } from "glass-ux-image/tools/retouch/brush";
import { cloneStamp } from "glass-ux-image/tools/retouch/clone";

const photo = `${import.meta.env.BASE_URL}sample.png`;

/** The sample photo in a contained stage, with a panel beside it as in an editor window. */
function Stage({
  editor,
  canvas,
  children,
}: {
  editor: EditorSession;
  canvas?: ReactNode;
  children?: ReactNode;
}) {
  const { status } = useSessionState(editor);
  const ready = status === "ready";
  return (
    <ImageEditorProvider editor={editor}>
      <div className="flex h-120 w-full min-w-0 flex-col overflow-hidden rounded-surface border border-line">
        <div className="flex min-h-0 flex-1">
          <div className="relative min-w-0 flex-1">
            {ready && canvas ? (
              canvas
            ) : (
              <ImageCanvas aria-label="Sample photo" className="absolute inset-0" />
            )}
          </div>
          {children && (
            <div className="w-80 shrink-0 overflow-y-auto border-s border-line p-4">
              {ready ? (
                children
              ) : (
                <p role="status" className="type-caption text-fg-muted">
                  Opening sample…
                </p>
              )}
            </div>
          )}
        </div>
      </div>
    </ImageEditorProvider>
  );
}

function RetouchOverlaySample() {
  const editor = useImageEditor({ source: photo, tools: [brush] });
  return (
    <Stage
      editor={editor}
      canvas={
        <ImageCanvas aria-label="Paint on the sample photo" className="absolute inset-0">
          <RetouchOverlay kind="brush" />
        </ImageCanvas>
      }
    >
      <RetouchPanel kind="brush" />
    </Stage>
  );
}

function RetouchPanelSample() {
  const editor = useImageEditor({ source: photo, tools: [cloneStamp] });
  return (
    <Stage editor={editor}>
      <RetouchPanel kind="clone" />
    </Stage>
  );
}

export default function Section() {
  return (
    <SectionPage>
      <Specimen
        title="Paint on the photo"
        description="The overlay turns pointer strokes into brush dabs, previewed at most once a frame and committed when the stroke ends. Each stroke is one undoable step."
      >
        <div data-live-demo="retouch-overlay" className="w-full">
          <RetouchOverlaySample />
        </div>
      </Specimen>
      <Specimen
        title="Brush settings"
        description="The shared brush controls, plus the settings one kind of retouching needs: here, the clone stamp's source."
      >
        <div data-live-demo="retouch-panel" className="w-full">
          <RetouchPanelSample />
        </div>
      </Specimen>
    </SectionPage>
  );
}
