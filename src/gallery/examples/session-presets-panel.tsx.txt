import { type ReactNode } from "react";
import { SectionPage, Specimen } from "glass-ux-docs-kit/react";
import { ImageCanvas } from "glass-ux-image/canvas";
import { ImageEditorProvider, useImageEditor, useSessionState } from "glass-ux-image/react";
import type { EditorSession } from "glass-ux-image/session";
import { SessionPresetsPanel, type ImagePreset } from "glass-ux-image/panels/presets";
import { develop } from "glass-ux-image/tools/develop";

const photo = `${import.meta.env.BASE_URL}sample.png`;

/** The sample photo in a contained stage, with a panel beside it as in an editor window. */
function Stage({
  editor,
  canvas,
  children,
}: {
  editor: EditorSession;
  canvas?: ReactNode;
  children?: ReactNode;
}) {
  const { status } = useSessionState(editor);
  const ready = status === "ready";
  return (
    <ImageEditorProvider editor={editor}>
      <div className="flex h-120 w-full min-w-0 flex-col overflow-hidden rounded-surface border border-line">
        <div className="flex min-h-0 flex-1">
          <div className="relative min-w-0 flex-1">
            {ready && canvas ? (
              canvas
            ) : (
              <ImageCanvas aria-label="Sample photo" className="absolute inset-0" />
            )}
          </div>
          {children && (
            <div className="w-80 shrink-0 overflow-y-auto border-s border-line p-4">
              {ready ? (
                children
              ) : (
                <p role="status" className="type-caption text-fg-muted">
                  Opening sample…
                </p>
              )}
            </div>
          )}
        </div>
      </div>
    </ImageEditorProvider>
  );
}

// Each preset is a patch over the recipe; develop settings merge with what is already there.
const presets: ImagePreset[] = [
  {
    id: "warm",
    label: "Golden hour",
    change: { filters: { develop: { temperature: 35, exposure: 0.2 } } },
  },
  {
    id: "cool",
    label: "Blue hour",
    change: { filters: { develop: { temperature: -30, tint: 8 } } },
  },
  { id: "punch", label: "Vivid", change: { filters: { develop: { vibrance: 40, contrast: 20 } } } },
  {
    id: "mono",
    label: "Black and white",
    change: { filters: { develop: { saturation: -100, contrast: 25 } } },
  },
  { id: "matte", label: "Matte", change: { filters: { develop: { blacks: 30, contrast: -15 } } } },
];

function SessionPresetsPanelSample() {
  const editor = useImageEditor({ source: photo, tools: [develop] });
  return (
    <Stage editor={editor}>
      <SessionPresetsPanel presets={presets} />
    </Stage>
  );
}

export default function Section() {
  return (
    <SectionPage>
      <Specimen
        title="Apply to the open picture"
        description="Choosing a preset commits it as one undoable step, named after the preset, through an action that onAction can refuse."
      >
        <div data-live-demo="session-presets-panel" className="w-full">
          <SessionPresetsPanelSample />
        </div>
      </Specimen>
    </SectionPage>
  );
}
